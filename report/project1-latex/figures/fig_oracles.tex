\OracleCol{0}{a}{\faCalculator}{CODDTest}{folds an expression into its value}
 {same tables t1, vt0 and view v0\\[1pt]{\rmfamily\itshape\fontsize{6.6}{7.6}\selectfont auxiliary query: EXISTS(\ldots) is false, so it folds to 0}}
 {{\ctitle\color{cblue}original query}\\SELECT vt0.c2 FROM t1 CROSS\\JOIN v0 ON (EXISTS (SELECT\\ v0.c0 FROM v0 WHERE false))\\FULL OUTER JOIN vt0 ON 1;}
 {\{-1\}}
 {{\ctitle\color{vartext}folded query}\\SELECT vt0.c2 FROM t1 CROSS\\JOIN v0 ON (0)\\FULL OUTER JOIN vt0 ON 1;}
\Outcome{a}{\{\,\}}{\{-1\}\mneq\{\,\}}{SQLite bug (CODDTest, Listing 8)}

\OracleCol{4.55}{b}{\faHistory}{DDLCheck}{rebuilds the schema without its history}
 {same final schema t1(c1 INT), then\\INSERT INTO t1 (c1) VALUES (0);}
 {{\ctitle\color{cblue}evolved schema}\\CREATE TABLE t1 (c1 INT, c2 BLOB)\\\ \ ROW\_FORMAT=REDUNDANT;\\ALTER TABLE t1 DROP c2;}
 {error}
 {{\ctitle\color{vartext}created directly}\\CREATE TABLE t1 (c1 INT)\\\ \ ROW\_FORMAT=REDUNDANT;}
\Outcome{b}{\{0\}}{error\mneq\{0\}}{MariaDB MDEV-35122}

\OracleCol{9.1}{c}{\faTags}{Radar}{drops optional metadata}

 {\{0\}}
 {{\ctitle\color{vartext}raw copy, no index}\\CREATE TABLE t1 (c1\\\ \ DOUBLE ZEROFILL);\\INSERT INTO t1 VALUES (0);}
\Outcome{c}{\{\,\}}{\{0\}\mneq\{\,\}}{MySQL \#110125}
